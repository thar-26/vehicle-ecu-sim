# Normal startup: ignition on, both speed sensors track each other
# closely throughout (small, realistic step changes -- speed doesn't
# teleport), clean shutdown. No faults expected.
EVENT IgnitionOn
SIGNAL vehicle_speed_a 0.0 0
SIGNAL vehicle_speed_b 0.0 0
EVENT InitializationComplete
KICK 100
SIGNAL vehicle_speed_a 3.0 300
SIGNAL vehicle_speed_b 3.0 300
TIME 400
KICK 500
SIGNAL vehicle_speed_a 6.0 600
SIGNAL vehicle_speed_b 6.5 600
TIME 700
KICK 800
SIGNAL vehicle_speed_a 8.0 900
SIGNAL vehicle_speed_b 8.5 900
TIME 950
EVENT ShutdownRequest
EVENT ShutdownComplete
